# ═══════════════════════════════════════════════════════════════════
# Phase A1.1 · Entity static metadata authority — cross-system regression probe
#
# Verifies (real orbital_sim entry path: main_menu → v1_hud → ShellBinder → GameSession):
#   1) scenario metadata is retrievable by entity id from the runtime authority:
#      side / identity / batch / ship_class / rcs_m2 / ir_signature / radius_km /
#      has_pd / facility_id / transponder_profile;
#   2) ballistics/PD state reads those values from the authority (no silent defaults);
#   3) contacts display never defaults missing info to "friend" — unknown stays unknown;
#   4) world-internal truth is NOT auto-copied into player tracks/display:
#      catalog truth side/identity must not appear in track/contact rows without evidence.
#
# Red-first evidence: on the pre-A1.1 implementation `session.entity_metadata()` does
# not exist and this probe prints the actual fallback/wrong values via `[A1] FAIL`.
#
# Output: `A1 ENTITY METADATA AUTHORITY PROBE RESULT: PASS=n FAIL=m`; quit 0/1.
# No rendering, no writes outside res:// reads + process env restoration.
# ═══════════════════════════════════════════════════════════════════

const TARGET_PROFILE := "orbital_sim"
const TARGET_SCENARIO := "res://framework/content/scenario_orbital_sim.json"
const MAIN_MENU := "res://main_menu.tscn"
const V1_HUD := "res://v1_hud.tscn"
const HUD_SCRIPT := "res://scripts/ui/v1_hud.gd"
const HUD_NODE := "V1HUD"

const MAX_BOOT_FRAMES := 600
const MAX_BOOT_MS := 60000
const MAX_BINDER_FRAMES := 120
const WATCHDOG_S := 180.0

const ENV_KEYS := ["DSH_BUILD", "DSH_SHOT", "DSH_NO_AUTOSKIP", "DSH_FLOW_SMOKE",
	"DSH_V1_TEST", "DSH_V1_SAVE_TEST", "DSH_LOAD_SLOT", "DSH_HUD_SIZE",
	"DSH_MAIN_MENU_VIEW", "DSH_SETTINGS_TEST", "DSH_PLAYER_NAME"]

const META_FIELDS := ["side", "identity", "batch", "ship_class", "rcs_m2",
	"ir_signature", "radius_km", "has_pd", "facility_id", "transponder_profile"]
const NUM_FIELDS := ["rcs_m2", "ir_signature", "radius_km"]

# A1.1 display contract (player-visible fields; must never contain world truth).
const TRACK_DISPLAY_NAME := "UNKNOWN TRACK"
const ENTITY_DISPLAY_NAME := "UNKNOWN CONTACT"
const ENTITY_DISPLAY_CLASS := "UNKNOWN"
const SIGNAL_DISPLAY_NAME := "UNKNOWN SIGNAL"
const SIGNAL_DISPLAY_CLASS := "SIGNAL"

var _pass := 0
var _fail := 0
var _failures: Array[String] = []
var _env_backup: Dictionary = {}
var _finished := false
var _hud = null
var _binder = null
var _session = null
var _truth: Dictionary = {}
var _truth_ids: Array[String] = []
var _main_menu_instantiated := false


func _initialize() -> void:
	print("════════════════════════════════════════════════════════════════")
	print("A1 ENTITY METADATA AUTHORITY PROBE · real entry main_menu → v1_hud → ShellBinder")
	print("Godot ", Engine.get_version_info().get("string", "?"), " · profile=", TARGET_PROFILE)
	print("════════════════════════════════════════════════════════════════")
	_setup_env()
	node_added.connect(_on_node_added)
	_watchdog()
	_run()


## Real-entry proof: the menu node is instantiated before auto-skip replaces it on the
## same frame, so record it via SceneTree.node_added instead of polling current_scene.
func _on_node_added(node: Node) -> void:
	if node == null:
		return
	var scr = node.get_script()
	if scr != null and String(scr.resource_path) == "res://scripts/ui/main_menu.gd":
		_main_menu_instantiated = true


func _watchdog() -> void:
	await create_timer(WATCHDOG_S).timeout
	if _finished:
		return
	print("[A1] FAIL watchdog timeout %.0fs" % WATCHDOG_S)
	_fail += 1
	_failures.append("watchdog timeout")
	_finish()


func _setup_env() -> void:
	for k in ENV_KEYS:
		_env_backup[k] = OS.get_environment(k)
	OS.set_environment("DSH_BUILD", TARGET_PROFILE)
	for k in ["DSH_SHOT", "DSH_NO_AUTOSKIP", "DSH_FLOW_SMOKE", "DSH_V1_TEST",
			"DSH_V1_SAVE_TEST", "DSH_LOAD_SLOT", "DSH_HUD_SIZE",
			"DSH_MAIN_MENU_VIEW", "DSH_SETTINGS_TEST", "DSH_PLAYER_NAME"]:
		OS.set_environment(k, "")


func _restore_env() -> void:
	for k in ENV_KEYS:
		var v := String(_env_backup.get(k, ""))
		if v == "":
			OS.unset_environment(k)
		else:
			OS.set_environment(k, v)


func _finish() -> void:
	if _finished:
		return
	_finished = true
	print("────────────────────────────────────────────────────────────────")
	print("A1 ENTITY METADATA AUTHORITY PROBE RESULT: PASS=%d FAIL=%d" % [_pass, _fail])
	if _fail > 0:
		for f in _failures:
			print("[A1] FAILED: ", f)
		print("A1 ENTITY METADATA AUTHORITY PROBE FAILED")
	else:
		print("A1 ENTITY METADATA AUTHORITY PROBE OK")
	_restore_env()
	quit(0 if _fail == 0 else 1)


func _ok(cond: bool, label: String, detail := "") -> void:
	if cond:
		_pass += 1
		print("[A1] PASS ", label, ("  " + detail) if detail != "" else "")
	else:
		_fail += 1
		var line := label + (("  (" + detail + ")") if detail != "" else "")
		_failures.append(line)
		print("[A1] FAIL ", line)


func _eq(actual, expected, label: String) -> void:
	_ok(actual == expected, label, "actual=%s expected=%s" % [str(actual), str(expected)])


func _step(title: String) -> void:
	print("")
	print("[A1] STEP ", title)


func _info(text: String) -> void:
	print("[A1] INFO ", text)


func _read_json(path: String) -> Dictionary:
	if not FileAccess.file_exists(path):
		return {}
	var f := FileAccess.open(path, FileAccess.READ)
	if f == null:
		return {}
	var parsed = JSON.parse_string(f.get_as_text())
	f.close()
	return parsed if parsed is Dictionary else {}


# ── boot ─────────────────────────────────────────────────────────

func _run() -> void:
	_step("1. real entry main_menu → v1_hud (change_scene_to_file)")
	var err := change_scene_to_file(MAIN_MENU)
	_ok(err == OK, "change_scene_to_file(main_menu) == OK", "err=%d" % err)
	if err != OK:
		_finish()
		return
	var boot := await _await_v1_hud()
	if boot["hud"] == null:
		_finish()
		return
	_hud = boot["hud"]
	_ok(bool(boot["saw_main_menu"]), "real entry passed main_menu",
		"saw=%s frames=%d" % [str(boot["saw_main_menu"]), int(boot["frames"])])
	_ok(bool(boot["scene_script_ok"]), "current_scene is V1HUD with v1_hud.gd",
		String(_hud.scene_file_path))
	_binder = _hud.get("binder")
	_session = _binder.get("session") if _binder != null else null
	_ok(_binder != null and bool(_binder.get("ready")), "ShellBinder ready",
		"errors=" + str(_binder.get("errors")) if _binder != null else "binder=null")
	_ok(_session != null, "GameSession available")
	if _session == null:
		_finish()
		return

	_step("2. scenario truth table (read directly from scenario JSON)")
	var sc := _read_json(TARGET_SCENARIO)
	_ok(not sc.is_empty(), "scenario JSON readable", TARGET_SCENARIO)
	var ents: Array = sc.get("entities", []) if sc.get("entities") is Array else []
	_eq(ents.size(), 7, "scenario entity count")
	for raw in ents:
		if raw is Dictionary:
			var e: Dictionary = raw
			var eid := String(e.get("id", ""))
			if eid != "":
				_truth[eid] = e.duplicate(true)
				_truth_ids.append(eid)
	_truth_ids.sort()
	_ok(_truth.size() == 7, "scenario truth table has 7 entities", str(_truth_ids))

	_step("3. entity metadata authority (catalog) by entity id")
	_ok(_session.has_method("entity_metadata"), "GameSession.entity_metadata(id) API exists")
	for eid in _truth_ids:
		var truth: Dictionary = _truth[eid]
		var meta: Dictionary = _get_metadata(eid)
		_ok(bool(meta.get("found", false)), "metadata found: " + eid,
			"meta_keys=" + str(meta.keys()))
		for field in META_FIELDS:
			_check_meta_field(eid, meta, field, truth.get(field))
	# unknown id must not invent metadata
	if _session.has_method("entity_metadata"):
		var unknown: Dictionary = _session.entity_metadata("__no_such_entity__")
		_ok(not bool(unknown.get("found", true)), "unknown id -> found=false",
			"meta=" + str(unknown))

	_step("4. ballistics / PD state reads authority (no silent defaults)")
	var states: Dictionary = {}
	if _session.has_method("_ballistics_states"):
		states = _session._ballistics_states()
	_ok(not states.is_empty(), "session._ballistics_states() non-empty", "n=" + str(states.size()))
	for eid in _truth_ids:
		var truth: Dictionary = _truth[eid]
		var st: Dictionary = states.get(eid, {}) if states.get(eid) is Dictionary else {}
		_ok(not st.is_empty(), "ballistics state exists: " + eid)
		if st.is_empty():
			continue
		_check_state_field(eid, st, "side", truth.get("side"))
		_check_state_field(eid, st, "has_pd", truth.get("has_pd"))
		_check_state_field(eid, st, "rcs_m2", truth.get("rcs_m2"), true)
		_check_state_field(eid, st, "ir_signature", truth.get("ir_signature"), true)
		_check_state_field(eid, st, "radius_km", truth.get("radius_km"), true)
		_ok(bool(st.get("metadata_found", false)),
			"ballistics state metadata_found=true: " + eid,
			"metadata_found=" + str(st.get("metadata_found", "missing")))
		_eq(String(st.get("radius_source", "missing")), "catalog",
			"ballistics radius from catalog (not silent default): " + eid)

	_step("5. contacts: missing info stays UNKNOWN; no truth auto-copy; no friend default")
	var contacts: Array = _binder.get("contacts") if _binder.get("contacts") is Array else []
	# NEGATIVE CONTROL (T1): 模拟生产回归——丢掉一个已获证据目标的 contact 行
	var nc_filtered: Array = []
	var nc_dropped := false
	for raw_nc in contacts:
		if not nc_dropped and raw_nc is Dictionary \
				and String((raw_nc as Dictionary).get("track_id", "")) != "":
			nc_dropped = true
			continue
		nc_filtered.append(raw_nc)
	contacts = nc_filtered
	var by_eid: Dictionary = {}
	for raw_c in contacts:
		if raw_c is Dictionary:
			var c: Dictionary = raw_c
			var ceid := String(c.get("entity_id", ""))
			if ceid != "":
				by_eid[ceid] = c
	_ok(contacts.size() > 0, "binder.contacts non-empty", "n=" + str(contacts.size()))
	# T1：从可信 owner 的已获证据导出「必须可见」集合，逐项要求 contact 存在；
	# 已知目标掉行必须变红（不能只靠"非空"或对缺失项 continue 跳过）。
	var trusted_owner := String(_binder.call("_intel_owner")) \
		if _binder != null and _binder.has_method("_intel_owner") else ""
	var owner_view: Dictionary = _session.intel_view(trusted_owner) \
		if _session != null and _session.has_method("intel_view") else {}
	var observed_tids: Array = []
	for raw_ot in (owner_view.get("tracks", []) as Array):
		if raw_ot is Dictionary:
			var otid := String((raw_ot as Dictionary).get("id", ""))
			if otid != "":
				observed_tids.append(otid)
	_ok(observed_tids.size() > 0, "trusted owner has acquired-evidence tracks",
		str(observed_tids))
	var missing_observed: Array = []
	for tid_ot in observed_tids:
		var found_row := false
		for raw_cc in contacts:
			if raw_cc is Dictionary \
					and String((raw_cc as Dictionary).get("track_id", "")) == tid_ot:
				found_row = true
				break
		if not found_row:
			missing_observed.append(String(tid_ot))
	_ok(missing_observed.is_empty(),
		"every acquired-evidence target has a contact row",
		str(missing_observed))
	for eid in _truth_ids:
		var truth: Dictionary = _truth[eid]
		var truth_side := String(truth.get("side", "")).to_lower()
		var truth_identity := String(truth.get("identity", "")).to_lower()
		var row: Dictionary = by_eid.get(eid, {}) if by_eid.get(eid) is Dictionary else {}
		if row.is_empty():
			# own ship / entity not exposed as contact is allowed; log for review
			_info("contact absent for " + eid + " (allowed only if not a player contact)")
			continue
		var badge := String(row.get("badge", ""))
		var cside := String(row.get("side", "missing")).to_lower()
		var cident := String(row.get("identity", "missing")).to_lower()
		var ctype := String(row.get("type", "missing"))
		var ssrc := String(row.get("side_source", "missing"))
		var isrc := String(row.get("identity_source", "missing"))
		_ok(ssrc != "missing", "contact has explicit side_source: " + eid,
			"badge=%s side=%s" % [badge, cside])
		_ok(isrc != "missing", "contact has explicit identity_source: " + eid,
			"badge=%s identity=%s" % [badge, cident])
		_check_display_fields(eid, row, truth)
		_check_hud_texts(eid, row, truth)
		if ssrc == "facility_registry":
			# facility ownership is content/database data, not world truth; identity must stay player-derived
			_ok(cident == "unknown", "facility contact identity stays unknown: " + eid,
				"identity=%s identity_source=%s" % [cident, isrc])
			continue
		# Default at boot: no player evidence → unknown, never "friend", never truth side/identity.
		_ok(cside == "unknown", "contact side unknown (no evidence): " + eid,
			"side=%s side_source=%s badge=%s" % [cside, ssrc, badge])
		if isrc == "track":
			_ok(cident == "unknown", "track identity unknown at boot: " + eid,
				"identity=%s" % cident)
		else:
			_eq(cident, "unknown", "contact identity unknown (no evidence): " + eid)
		_ok(cside != "friend", "contact does not default to friend: " + eid, "side=" + cside)
		if truth_side != "":
			_ok(cside != truth_side,
				"contact side not auto-copied from world truth: " + eid,
				"contact=%s truth=%s" % [cside, truth_side])
		if isrc == "track":
			_ok(cident != truth_identity,
				"track identity not auto-copied from world truth: " + eid,
				"contact=%s truth=%s" % [cident, truth_identity])
		if badge == "ENTITY":
			_ok(ctype != "friendly", "entity contact not defaulted to friendly type: " + eid,
				"type=" + ctype)
	# global guards
	var friend_contacts: Array = []
	var truth_leaks: Array = []
	for eid2 in by_eid.keys():
		var c2: Dictionary = by_eid[eid2]
		if String(c2.get("side_source", "")) == "facility_registry":
			continue
		var sid := String(c2.get("side", "")).to_lower()
		if sid == "friend":
			friend_contacts.append(String(eid2))
		var t2: Dictionary = _truth.get(String(eid2), {})
		if not t2.is_empty() and sid != "" and sid == String(t2.get("side", "")).to_lower():
			truth_leaks.append("%s(%s)" % [String(eid2), sid])
	_ok(friend_contacts.is_empty(), "no silent friend default in contacts",
		"friend_contacts=" + str(friend_contacts))
	_ok(truth_leaks.is_empty(), "no truth side leaked into contacts",
		"leaks=" + str(truth_leaks))

	_step("5b. SIGNAL rows: truth-like target metadata must stay anonymous")
	var sig_truth := {"id": "sig_probe_entity", "name": "TRUTH-SIGNAL-NAME",
		"batch": "TRUTH-BATCH-9", "ship_class": "TRUTH-CLASS",
		"side": "hostile", "identity": "hostile"}
	var up: Dictionary = _session.signal_upsert_target({
		"id": "sig_probe_target", "entity_id": "sig_probe_entity",
		"name": sig_truth["name"], "batch": sig_truth["batch"],
		"ship_class": sig_truth["ship_class"], "side": sig_truth["side"],
		"identity": sig_truth["identity"], "pos_km": [0.0, 0.0, 0.0],
		"emissions": [], "light_capable": true, "receiver_facing": true})
	_ok(bool(up.get("ok", false)), "synthetic signal target registered",
		str(up.get("reason", "")))
	_binder.refresh()
	var sig_row: Dictionary = {}
	var sig_contacts: Array = _binder.get("contacts") if _binder.get("contacts") is Array else []
	for raw_s in sig_contacts:
		if raw_s is Dictionary and String((raw_s as Dictionary).get("entity_id", "")) == "sig_probe_entity":
			sig_row = raw_s
			break
	# JC-A01/R3b：signal 注册本身不是「该 owner 收到观测」，不得产生玩家可见行。
	_ok(sig_row.is_empty(),
		"signal registration alone does not create a player contact",
		str(sig_row.keys()))
	if not sig_row.is_empty():
		# 若产品的合法到达链确实开放了该行，仍必须匿名且无真值。
		_ok(String(sig_row.get("side", "")).to_lower() == "unknown",
			"SIGNAL row side unknown (no evidence)",
			"side=" + String(sig_row.get("side", "")))
		_ok(String(sig_row.get("identity", "")).to_lower() == "unknown",
			"SIGNAL row identity unknown (no evidence)",
			"identity=" + String(sig_row.get("identity", "")))
		_check_display_fields("sig_probe_entity", sig_row, sig_truth)
		_check_hud_texts("sig_probe_entity", sig_row, sig_truth)
	# cleanup (do not leak the synthetic target into later assertions)
	_session.signal_remove_target("sig_probe_target")
	_session.signal_remove_target("sig_probe_entity")
	_binder.refresh()

	_step("6. truth vs player view separation (explicit evidence)")
	# The simulation authority knows harmattan is hostile; the player-facing track/contact does not.
	var hmeta := _get_metadata("harmattan")
	var hrow: Dictionary = by_eid.get("harmattan", {})
	_eq(String(hmeta.get("side", "")), "hostile", "catalog truth harmattan side")
	if not hrow.is_empty():
		_ok(String(hrow.get("side", "")).to_lower() != String(hmeta.get("side", "")).to_lower(),
			"player contact does not reveal catalog truth side",
			"contact=%s truth=%s" % [String(hrow.get("side", "")), String(hmeta.get("side", ""))])
		_ok(String(hrow.get("identity", "")).to_lower() != String(hmeta.get("identity", "")).to_lower(),
			"player track identity is evidence-derived, not catalog truth",
			"contact=%s truth=%s" % [String(hrow.get("identity", "")), String(hmeta.get("identity", ""))])

	_step("7. event stream texts (COMMS/COMBAT/traffic/bottom) + internal refs preserved")
	_check_event_stream_texts()
	_finish()


func _check_event_stream_texts() -> void:
	# Synthetic COMMS event with sentinel internal ids/names.
	var cev := {"from_id": "SENTINEL-C-FROM", "to_id": "SENTINEL-C-TO",
		"from_name": "SENTINEL-C-FROM", "to_name": "SENTINEL-C-TO",
		"from": "SENTINEL-C-FROM", "to": "SENTINEL-C-TO", "mode": "LD",
		"text": "[SENTINEL-C-FROM 至 SENTINEL-C-TO]-[LD]-[CONTENT]", "kind": "SENT"}
	var cbefore := JSON.stringify(cev)
	var ctext: Dictionary = _hud.call("hud_comms_event_text", cev)
	var cvis := "%s|%s|%s" % [String(ctext.get("from", "")), String(ctext.get("to", "")),
		String(ctext.get("text", ""))]
	_assert_event_text("comms synthetic", cvis,
		["SENTINEL-C-FROM", "SENTINEL-C-TO"], ["UNKNOWN"])
	_eq(JSON.stringify(cev), cbefore, "comms synthetic event unchanged by formatting")

	# Synthetic COMMS event using real internal ids (must show safe labels, not ids).
	var kev := {"from_id": "harmattan", "to_id": "raider_2",
		"from_name": "harmattan", "to_name": "raider_2",
		"from": "harmattan", "to": "raider_2", "mode": "LD",
		"text": "[harmattan 至 raider_2]-[LD]-[CONTENT]", "kind": "SENT"}
	var kbefore := JSON.stringify(kev)
	var ktext: Dictionary = _hud.call("hud_comms_event_text", kev)
	var kvis := "%s|%s|%s" % [String(ktext.get("from", "")), String(ktext.get("to", "")),
		String(ktext.get("text", ""))]
	_assert_event_text("comms known internal", kvis, ["harmattan", "raider_2"],
		["t0001", "unknown track", "unknown contact", "unknown"])
	_eq(JSON.stringify(kev), kbefore, "comms known event unchanged by formatting")

	# COMMS link endpoints.
	var lev := {"a": "SENTINEL-LINK-A", "b": "SENTINEL-LINK-B", "mode": "LD"}
	var lt := String(_hud.call("hud_comms_link_text", lev))
	_assert_event_text("comms link", lt, ["SENTINEL-LINK-A", "SENTINEL-LINK-B"], ["UNKNOWN"])

	# Synthetic COMBAT event payload with sentinel internal refs (incl. nested).
	var pev := {"t": 1.0, "kind": "weapon_launch", "payload": {
		"target_id": "SENTINEL-T", "threat_id": "SENTINEL-H",
		"shooter_id": "SENTINEL-S", "entity_id": "SENTINEL-E",
		"nested": {"target_id": "SENTINEL-N"}}}
	var pbefore := JSON.stringify(pev)
	var pt := String(_hud.call("hud_combat_event_text", pev))
	_assert_event_text("combat synthetic", pt,
		["SENTINEL-T", "SENTINEL-H", "SENTINEL-S", "SENTINEL-E", "SENTINEL-N"], ["UNKNOWN"])
	_eq(JSON.stringify(pev), pbefore, "combat synthetic event unchanged by formatting")

	# COMBAT event with real internal refs.
	var pkev := {"t": 2.0, "kind": "weapon_launch", "payload": {
		"target_id": "harmattan", "shooter_id": "apsis"}}
	var pkbefore := JSON.stringify(pkev)
	var pkt := String(_hud.call("hud_combat_event_text", pkev))
	_assert_event_text("combat known internal", pkt, ["harmattan", "apsis"],
		["t0001", "1000", "own ship", "unknown"])
	_eq(JSON.stringify(pkev), pkbefore, "combat known event unchanged by formatting")

	# 真实 CombatService 产生的 incoming_impact（嵌套 damage_event.source_id = 内部威胁 ID）。
	# 事件结构由生产代码 combat_service._resolve_incoming() 生成，避免手写结构漂移；
	# 这是 HUD 战斗事件栏实际会格式化的真实事件形态。
	var cs_script = load("res://scripts/core/combat_service.gd")
	var cs = cs_script.new({}) if cs_script != null else null
	_ok(cs != null and bool(cs.ready), "real CombatService ready for incoming_impact fixture")
	if cs != null and bool(cs.ready):
		var incr: Dictionary = cs.incoming_missile("SENTINEL-THREAT", "SENTINEL-TARGET", 0.0,
			{"eta_s": 0.0, "interceptable": false})
		_ok(bool(incr.get("ok", false)), "real incoming missile fixture registered")
		cs.tick(0.0)
		var iev: Dictionary = {}
		for raw_ie in (cs.events as Array):
			if raw_ie is Dictionary and String((raw_ie as Dictionary).get("kind", "")) == "incoming_impact":
				iev = raw_ie
				break
		_ok(not iev.is_empty(), "real incoming_impact event emitted")
		if not iev.is_empty():
			var ipl: Dictionary = iev.get("payload", {}) if iev.get("payload") is Dictionary else {}
			var ide: Dictionary = ipl.get("damage_event", {}) if ipl.get("damage_event") is Dictionary else {}
			_eq(String(ide.get("schema", "")), "combat_damage_event.v1",
				"incoming_impact damage_event schema")
			_eq(String(ide.get("source_id", "")), "SENTINEL-THREAT",
				"incoming_impact damage_event.source_id is internal threat ref")
			_eq(String(ide.get("target_id", "")), "SENTINEL-TARGET",
				"incoming_impact damage_event.target_id is internal target ref")
			var src_before := String(ide.get("source_id", ""))
			var ibefore := JSON.stringify(iev)
			var itext := String(_hud.call("hud_combat_event_text", iev))
			_assert_event_text("combat incoming_impact", itext,
				["SENTINEL-THREAT", "SENTINEL-TARGET"], ["UNKNOWN"])
			_eq(JSON.stringify(iev), ibefore, "incoming_impact event unchanged by HUD formatting")
			_eq(String(ide.get("source_id", "")), src_before,
				"incoming_impact source_id internal value preserved after formatting")
			_ok(itext.find("INC-000001") >= 0,
				"incoming_impact event id kept visible (event id, not entity ref)",
				"text=" + itext)

	# Traffic payload JSON display.
	var tev := {"track_id": "SENTINEL-TRK", "entity_id": "SENTINEL-ENT",
		"nested": {"target_id": "SENTINEL-NEST"}}
	var tt := String(_hud.call("hud_event_payload_text", tev))
	_assert_event_text("traffic payload", tt,
		["SENTINEL-TRK", "SENTINEL-ENT", "SENTINEL-NEST"], ["UNKNOWN"])

	# Bottom event log label + generic visible text.
	var bev := {"type": "comms", "payload": {"label": "REPORT harmattan → raider_2"}}
	var bt := String(_hud.call("hud_bottom_event_text", bev))
	_assert_event_text("bottom event", bt, ["harmattan", "raider_2"],
		["t0001", "unknown"])
	var vt := String(_hud.call("hud_visible_text", "LOG harmattan → raider_2"))
	_assert_event_text("visible text", vt, ["harmattan", "raider_2"],
		["t0001", "unknown"])

	# Real COMMS route: service stores internal ids; HUD formatting must not mutate them.
	var nodes: Dictionary = _session.comms.nodes if _session != null and _session.comms != null else {}
	var ids: Array = nodes.keys()
	var route_from := ""
	var route_to := ""
	var sent: Dictionary = {}
	for i in range(ids.size()):
		for j in range(ids.size()):
			if i == j:
				continue
			var r: Dictionary = _session.send_comms(String(ids[i]), String(ids[j]), "LD",
				"COMMS_SENT", {}, {})
			if bool(r.get("ok", false)):
				sent = r
				route_from = String(ids[i])
				route_to = String(ids[j])
				break
		if not sent.is_empty():
			break
	_ok(not sent.is_empty(), "real comms route send ok",
		"nodes=" + str(ids.size()))
	if not sent.is_empty():
		var evs: Array = _session.comms.vm().get("events", []) if _session.comms.vm().get("events") is Array else []
		var rev: Dictionary = {}
		for raw_e in evs:
			if raw_e is Dictionary and String((raw_e as Dictionary).get("delivery_id", "")) == String(sent.get("delivery_id", "")):
				rev = (raw_e as Dictionary).duplicate(true)
				break
		_ok(not rev.is_empty(), "real comms event found")
		if not rev.is_empty():
			_eq(String(rev.get("from", "")), route_from, "comms event from internal id preserved")
			_eq(String(rev.get("to", "")), route_to, "comms event to internal id preserved")
			var rbefore := JSON.stringify(rev)
			var rdisp: Dictionary = _hud.call("hud_comms_event_text", rev)
			var rvis := "%s|%s|%s" % [String(rdisp.get("from", "")), String(rdisp.get("to", "")),
				String(rdisp.get("text", ""))]
			_assert_event_text("real comms event", rvis, [route_from, route_to],
				["unknown", "t0001", "1000", "own ship"])
			_eq(String(rev.get("from", "")), route_from, "comms event from unchanged after formatting")
			_eq(String(rev.get("to", "")), route_to, "comms event to unchanged after formatting")
			_eq(JSON.stringify(rev), rbefore, "comms stored event payload unchanged by HUD formatting")

	# Combat target internal ref unchanged after formatting.
	var tracks: Array = _binder.track_vm.get("tracks", []) if _binder != null and _binder.track_vm.get("tracks") is Array else []
	var tid := ""
	for raw_t in tracks:
		if raw_t is Dictionary and String((raw_t as Dictionary).get("track_id", "")) != "":
			tid = String((raw_t as Dictionary).get("track_id", ""))
			break
	if tid != "":
		_hud.set("_combat_target", tid)
		_hud.call("hud_combat_event_text", pev)
		_eq(String(_hud.get("_combat_target")), tid, "combat target internal id preserved after HUD formatting")
	else:
		_ok(false, "combat target track id available")


func _assert_event_text(label: String, text: String, forbidden: Array, required: Array) -> void:
	var low := text.to_lower()
	for f in forbidden:
		_ok(low.find(String(f).to_lower()) < 0, label + " no internal token: " + String(f),
			"text=" + text)
	var found := false
	for r in required:
		if low.find(String(r).to_lower()) >= 0:
			found = true
			break
	_ok(found, label + " contains safe/placeholder label",
		"text=" + text + " required=" + str(required))


func _check_hud_texts(eid: String, row: Dictionary, truth: Dictionary) -> void:
	if _hud == null:
		_ok(false, "HUD node available for text checks: " + eid)
		return
	var contacts: Array = _binder.get("contacts") if _binder != null and _binder.get("contacts") is Array else []
	_hud.set("_contacts", contacts.duplicate(true))
	_hud.call("_normalize_contacts")
	var hcontacts: Array = _hud.get("_contacts") if _hud.get("_contacts") is Array else []
	var idx := -1
	var hrow: Dictionary = {}
	for i in range(hcontacts.size()):
		if hcontacts[i] is Dictionary and String((hcontacts[i] as Dictionary).get("entity_id", "")) == eid:
			idx = i
			hrow = (hcontacts[i] as Dictionary).duplicate(true)
			break
	_ok(idx >= 0, "HUD contact row exists: " + eid)
	if idx < 0:
		return
	_hud.set("_selected", idx)
	var internal := String(hrow.get("entity_id", ""))
	var disp: Dictionary = _hud.call("_contact_display", hrow)
	var facility_ex := String(disp.get("source", "")) == "facility_registry"

	# Actual COMMS action text (not only a helper).
	_hud.set("_comms_target", internal)
	_hud.call("_comms_action", "HOLD")
	_assert_hud_text(eid, "comms_msg", String(_hud.get("_comms_msg")), truth, disp, facility_ex)
	_eq(String(_hud.get("_comms_target")), internal, "HUD comms internal target preserved: " + eid)

	# Actual IDENT action text (INTERROGATE result message).
	_hud.set("_ident_row", hrow.duplicate(true))
	_hud.call("_ident_action", "INTERROGATE")
	_assert_hud_text(eid, "ident_msg", String(_hud.get("_ident_msg")), truth, disp, facility_ex)
	var irow = _hud.get("_ident_row")
	if irow is Dictionary:
		_eq(String((irow as Dictionary).get("entity_id", "")), internal,
			"HUD ident internal target preserved: " + eid)

	# Actual OPTICAL action text (IFF message + transponder reply line).
	_hud.set("_optical_target", internal)
	_hud.call("_optical_action", "IFF")
	_assert_hud_text(eid, "optical_msg", String(_hud.get("_optical_msg")), truth, disp, facility_ex)
	var txp = _hud.get("_txp_msg")
	if txp != null and String(txp) != "":
		_assert_hud_text(eid, "txp_msg", String(txp), truth, disp, facility_ex)
	_eq(String(_hud.get("_optical_target")), internal, "HUD optical internal target preserved: " + eid)

	# Generated texts used by fusion log / command queue / combat terminal.
	if _hud.has_method("hud_target_texts"):
		var texts: Dictionary = _hud.call("hud_target_texts", hrow)
		for key in ["comms_header", "comms_type", "comms_send", "optical_entity",
				"optical_msg", "ident_interrogate", "ident_iff", "ident_clear",
				"fusion_target", "command_target", "combat_target"]:
			_assert_hud_text(eid, key, String(texts.get(key, "")), truth, disp, facility_ex)
		_eq(String(texts.get("internal_target", "")), internal,
			"HUD generated texts keep internal target: " + eid)
		_eq(String(texts.get("internal_track_id", "")), String(hrow.get("track_id", "")),
			"HUD generated texts keep internal track id: " + eid)
	else:
		_ok(false, "HUD hud_target_texts helper exists (actual text builder)")


func _assert_hud_text(eid: String, label: String, text: String, truth: Dictionary,
		disp: Dictionary, facility_ex: bool) -> void:
	_ok(text != "", "HUD %s non-empty: %s" % [label, eid], "text=" + text)
	_ok(not _contains_truth(text, truth, eid, not facility_ex),
		"HUD %s contains no truth: %s" % [label, eid], "text=" + text)
	var did := String(disp.get("id", ""))
	var dname := String(disp.get("name", ""))
	var uses_display := (did != "" and text.find(did) >= 0) or (dname != "" and text.find(dname) >= 0)
	_ok(uses_display, "HUD %s uses safe display: %s" % [label, eid],
		"text=%s disp=%s" % [text, str(disp)])


func _check_display_fields(eid: String, row: Dictionary, truth: Dictionary) -> void:
	var did := String(row.get("display_id", ""))
	var dname := String(row.get("display_name", ""))
	var dcls := String(row.get("display_class", ""))
	var dsrc := String(row.get("display_source", ""))
	var badge := String(row.get("badge", ""))
	_ok(did != "" and dname != "" and dcls != "" and dsrc != "",
		"display fields present: " + eid,
		"id=%s name=%s class=%s source=%s" % [did, dname, dcls, dsrc])
	# 内部引用必须保留：选择/传感器/交战仍用真实 entity_id，而不是显示 ID。
	_eq(String(row.get("entity_id", "")), eid, "internal entity_id preserved: " + eid)
	if dsrc == "facility_registry":
		# Facility registry public fields are an allowed, explicitly-sourced exception.
		var fid := String(row.get("facility_id", ""))
		var card: Dictionary = _session.facility_lookup(fid) if _session != null else {}
		var fac: Dictionary = card.get("facility", {}) if card.get("facility") is Dictionary else {}
		_ok(not fac.is_empty(), "facility registry record exists: " + eid, "facility_id=" + fid)
		_eq(dsrc, "facility_registry", "facility display source marked: " + eid)
		if not fac.is_empty():
			_eq(dname, String(fac.get("name", "")), "facility display name from registry: " + eid)
			_eq(dcls, String(fac.get("facility_class", "")), "facility display class from registry: " + eid)
			_ok(did == String(fac.get("facility_code", ""))
				or did == String(fac.get("transponder_code", "")),
				"facility display id from public registry code: " + eid,
				"id=%s code=%s txp=%s" % [did, String(fac.get("facility_code", "")),
					String(fac.get("transponder_code", ""))])
	else:
		if badge == "TRACK" or String(row.get("track_id", "")) != "":
			# JC-A01：正常 A01 路径的玩家航迹来源是 owner 情报视图（不是全局 track 表）。
			_eq(dsrc, "intel_view", "track display source: " + eid)
			# 显示 ID 必须是安全匿名标签：不得回读实体名/batch/class/side 等未获证据真值。
			_ok(did != "" and did != eid and not _contains_truth(did, _truth.get(eid, {}) if _truth.get(eid) is Dictionary else {}, eid, false),
				"track display id is anonymous safe label: " + eid, "id=" + did)
			# 追溯：owner 必须是 binder 的可信本舰 owner，且 track_id 能在该 owner 的
			# intel_view.tracks 找到，内容/采样时刻与该证据一致。
			var row_owner := String(row.get("intel_owner_id", ""))
			var binder_owner := String(_binder.call("_intel_owner")) \
				if _binder != null and _binder.has_method("_intel_owner") else ""
			_eq(row_owner, binder_owner, "contact owner is binder trusted owner: " + eid)
			var ov: Dictionary = _session.intel_view(row_owner) \
				if _session != null and _session.has_method("intel_view") else {}
			var tid_row := String(row.get("track_id", ""))
			var found_tr: Dictionary = {}
			for raw_tr in (ov.get("tracks", []) if ov.get("tracks") is Array else []):
				if raw_tr is Dictionary \
						and String((raw_tr as Dictionary).get("id", "")) == tid_row:
					found_tr = raw_tr
					break
			_ok(not found_tr.is_empty(),
				"contact track traceable in owner view: " + eid, tid_row)
			if not found_tr.is_empty():
				_ok(absf(float(found_tr.get("range_km", -1.0))
						- float(row.get("range_km", -2.0))) <= 1.0e-6,
					"contact range matches owner evidence: " + eid,
					"%s vs %s" % [str(found_tr.get("range_km", null)),
						str(row.get("range_km", null))])
				_ok(absf(float(found_tr.get("last_observed_t", -1.0))
						- float(row.get("observed_t", -2.0))) <= 1.0e-6,
					"contact observed_t matches owner evidence: " + eid,
					"%s vs %s" % [str(found_tr.get("last_observed_t", null)),
						str(row.get("observed_t", null))])
			_eq(dname, TRACK_DISPLAY_NAME, "track display name generic: " + eid)
			_ok(dcls.begins_with("TRACK L"), "track display class only level: " + eid,
				"class=" + dcls)
		elif dsrc == "ownship":
			_eq(did, "1000", "ownship display id: " + eid)
			_eq(dname, "OWN SHIP", "ownship display name: " + eid)
			_eq(dcls, "FLAGSHIP", "ownship display class: " + eid)
		elif badge == "SIGNAL":
			_eq(dsrc, "anonymous_signal", "signal display source: " + eid)
			_ok(did.begins_with("SIGNAL-"), "signal display id anonymous: " + eid,
				"id=" + did)
			_eq(dname, SIGNAL_DISPLAY_NAME, "signal display name generic: " + eid)
			_eq(dcls, SIGNAL_DISPLAY_CLASS, "signal display class generic: " + eid)
		else:
			_eq(dsrc, "anonymous", "entity display source anonymous: " + eid)
			_ok(did.begins_with("UNKNOWN-"), "entity display id anonymous: " + eid,
				"id=" + did)
			_eq(dname, ENTITY_DISPLAY_NAME, "entity display name generic: " + eid)
			_eq(dcls, ENTITY_DISPLAY_CLASS, "entity display class generic: " + eid)
		_ok(not _contains_truth(did, truth, eid)
			and not _contains_truth(dname, truth, eid)
			and not _contains_truth(dcls, truth, eid),
			"display fields contain no scenario truth: " + eid,
			"id=%s name=%s class=%s" % [did, dname, dcls])
	# Legacy display-oriented fields (other UI paths may still read them) must equal safe fields.
	_eq(String(row.get("id", "")), did, "legacy contact id equals safe display id: " + eid)
	_eq(String(row.get("name", "")), dname, "legacy contact name equals safe display name: " + eid)
	_eq(String(row.get("cls", "")), dcls, "legacy contact cls equals safe display class: " + eid)
	# HUD contract helper (used by v1_hud) must return the same safe fields.
	if _binder != null and _binder.has_method("contact_display"):
		var hd: Dictionary = _binder.contact_display(row)
		_eq(String(hd.get("id", "")), did, "HUD display id uses safe field: " + eid)
		_eq(String(hd.get("name", "")), dname, "HUD display name uses safe field: " + eid)
		_eq(String(hd.get("cls", "")), dcls, "HUD display class uses safe field: " + eid)
		_ok(not _contains_truth(String(hd.get("id", "")), truth, eid)
			and not _contains_truth(String(hd.get("name", "")), truth, eid)
			and not _contains_truth(String(hd.get("cls", "")), truth, eid),
			"HUD display fields contain no truth: " + eid)
	else:
		_ok(false, "ShellBinder.contact_display(row) exists (HUD contract)")


func _contains_truth(value: String, truth: Dictionary, eid: String, include_name := true) -> bool:
	var v := value.to_lower()
	var keys := ["batch", "ship_class", "side", "identity", "id"]
	if include_name:
		keys.append("name")
	for k in keys:
		var t := String(truth.get(k, "")).to_lower()
		# Generic "unknown"/"none" values are not identifying truth; UNKNOWN display is allowed.
		if t != "" and t not in ["unknown", "none", "pending"] and v.find(t) >= 0:
			return true
	if eid != "" and v.find(eid.to_lower()) >= 0:
		return true
	return false


func _get_metadata(eid: String) -> Dictionary:
	if _session != null and _session.has_method("entity_metadata"):
		var m = _session.entity_metadata(eid)
		if m is Dictionary:
			return m
	# pre-A1.1 fallback: only world/snapshot fields exist (missing static metadata)
	var out := {"found": false, "legacy_fallback": true, "id": eid}
	var snap: Dictionary = _session.snapshot() if _session != null else {}
	for raw in (snap.get("entities", []) if snap.get("entities", []) is Array else []):
		if raw is Dictionary and String((raw as Dictionary).get("id", "")) == eid:
			out = (raw as Dictionary).duplicate(true)
			out["found"] = true
			out["legacy_fallback"] = true
			break
	return out


func _check_meta_field(eid: String, meta: Dictionary, field: String, expected) -> void:
	# scenario JSON omits absent strings (null); canonical catalog records them as "".
	if expected == null:
		expected = ""
	var actual = meta.get(field, "<missing>")
	if expected is bool:
		# Type-aware: legacy fallback returns "<missing>" (String); never compare String==bool.
		_ok(actual is bool and actual == expected, "metadata %s.%s preserved" % [eid, field],
			"actual=%s expected=%s legacy=%s" % [str(actual), str(expected), str(meta.get("legacy_fallback", false))])
	elif field in NUM_FIELDS:
		var ok_num := false
		if actual is int or actual is float:
			ok_num = absf(float(actual) - float(expected)) <= 1.0e-9
		_ok(ok_num, "metadata %s.%s preserved" % [eid, field],
			"actual=%s expected=%s legacy=%s" % [str(actual), str(expected), str(meta.get("legacy_fallback", false))])
	else:
		_ok(actual == expected, "metadata %s.%s preserved" % [eid, field],
			"actual=%s expected=%s legacy=%s" % [str(actual), str(expected), str(meta.get("legacy_fallback", false))])


func _check_state_field(eid: String, st: Dictionary, field: String, expected, is_num := false) -> void:
	var actual = st.get(field, "<missing>")
	if expected is bool:
		_ok(actual is bool and actual == expected, "ballistics %s.%s from authority" % [eid, field],
			"actual=%s expected=%s" % [str(actual), str(expected)])
	elif is_num:
		var ok_num := false
		if actual is int or actual is float:
			ok_num = absf(float(actual) - float(expected)) <= 1.0e-9
		_ok(ok_num, "ballistics %s.%s from authority" % [eid, field],
			"actual=%s expected=%s" % [str(actual), str(expected)])
	else:
		_ok(actual == expected, "ballistics %s.%s from authority" % [eid, field],
			"actual=%s expected=%s" % [str(actual), str(expected)])


func _await_v1_hud() -> Dictionary:
	var frames := 0
	var saw_main_menu := false
	var scene_script_ok := false
	var t0 := Time.get_ticks_msec()
	while frames < MAX_BOOT_FRAMES and (Time.get_ticks_msec() - t0) < MAX_BOOT_MS:
		await process_frame
		frames += 1
		var cs = current_scene
		if cs == null:
			continue
		var path := String(cs.scene_file_path)
		if path == MAIN_MENU:
			saw_main_menu = true
			continue
		if path != V1_HUD:
			continue
		var hud = cs
		var f2 := 0
		while f2 < MAX_BINDER_FRAMES:
			var b = hud.get("binder")
			if b != null and bool(b.get("ready")):
				break
			await process_frame
			f2 += 1
		var scr = hud.get_script()
		var script_path := ""
		if scr != null:
			script_path = String(scr.resource_path)
		scene_script_ok = script_path == HUD_SCRIPT and String(hud.name) == HUD_NODE
		return {"hud": hud, "frames": frames,
			"saw_main_menu": saw_main_menu or _main_menu_instantiated,
			"scene_script_ok": scene_script_ok,
			"elapsed_ms": Time.get_ticks_msec() - t0}
	print("[A1] FAIL entry timeout waiting for v1_hud (%d frames / %d ms)" % [MAX_BOOT_FRAMES, MAX_BOOT_MS])
	_fail += 1
	_failures.append("entry timeout")
	return {"hud": null, "frames": frames,
		"saw_main_menu": saw_main_menu or _main_menu_instantiated,
		"scene_script_ok": false, "elapsed_ms": Time.get_ticks_msec() - t0}
